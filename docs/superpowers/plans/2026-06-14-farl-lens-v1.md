# FaRl Lens v1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Server-side financial observability for corren — 4 read-only aggregation endpoints (overview, flows, rollup, timeseries) — plus a Horizon "Lens" page consuming them (overview cards, rollup, time-series chart, polished animated flows table).

**Architecture:** New `analytics/` package (sibling of `sharia/`, `guard/`) holding result types + an `AnalyticsStore` interface, embedded into `storage.Store`, implemented by both flavors with `GROUP BY`/`SUM` SQL (server-side aggregation, never fetch-all). A `lens_controller.go` exposes `/:ledger/lens/...`. 100% read-only — no `Commit` hook, no writes. Frontend: a Lens page in Horizon on a branch off `feature/contracts-ui`.

**Tech Stack:** Go 1.16, go-sqlbuilder, gin, fx; React 16 + styled-components (Horizon). No new dependency.

**Spec:** `docs/superpowers/specs/2026-06-14-farl-lens-v1-design.md`
**Branches:** corren `feature/farl-lens-v1` (created, off `feature/farl-guard-v1`); Horizon branch off `feature/contracts-ui` (created in Task 8). NEVER commit to main.

**Execution order:** BACKEND first (Tasks 1–7, corren, fully testable), THEN the UI (Tasks 8–10, Horizon). The `/flows` contract is frozen by Task 4 + its tests before any UI consumes it.

**Import graph (no cycle):** `analytics` → `core` only. `storage` → `analytics` (+ sharia + guard + core). `ledger`/`api` → `analytics` via the store.

import (

// seed a known ledger: two transfers, distinct days, two assets.
func seedLens(t *testing.T, s *SQLiteStore) {
	t.Helper()
	// tx 0: @world -> @client:a  [DZD.2 1000] at 2026-07-01
	// tx 1: @client:a -> @bank:treasury [DZD.2 400] at 2026-07-02
	// tx 2: @world -> @client:a  [EUR.2 50]  at 2026-07-02
	txs := []core.Transaction{
		{ID: 0, Timestamp: "2026-07-01T10:00:00Z", Postings: []core.Posting{
			{Source: "@world", Destination: "@client:a", Amount: 1000, Asset: "DZD.2"}}},
		{ID: 1, Timestamp: "2026-07-02T10:00:00Z", Postings: []core.Posting{
			{Source: "@client:a", Destination: "@bank:treasury", Amount: 400, Asset: "DZD.2"}}},
		{ID: 2, Timestamp: "2026-07-02T11:00:00Z", Postings: []core.Posting{
			{Source: "@world", Destination: "@client:a", Amount: 50, Asset: "EUR.2"}}},
	}
	if err := s.SaveTransactions(txs); err != nil {
		t.Fatal(err)
	}
	// one contract (PROMISE) and one guard event (deny) for the rollup
	raw, _ := sharia.BuildSchedule(1000, 100, 2, "2026-07-01T00:00:00Z", 30)
	_ = raw
	if err := s.SaveContract(sharia.Contract{ID: "c1", Type: sharia.TypeMurabaha,
		State: sharia.StatePromise, Params: []byte("{}"),
		TemplateVersion: "x", CreatedAt: "2026-07-01T00:00:00Z", UpdatedAt: "2026-07-01T00:00:00Z"}); err != nil {
		t.Fatal(err)
	}
	if err := s.AppendGuardEvent(guard.GuardEvent{RuleID: "r1", Action: guard.ActionDeny,
		Reason: "x", StandardRef: "P", CreatedAt: "2026-07-02T00:00:00Z"}); err != nil {
		t.Fatal(err)
	}
}

func withLensStore(t *testing.T, f func(s *SQLiteStore)) {
	t.Helper()
	s, err := NewStore("lenstest")
	if err != nil {
		t.Fatal(err)
	}
	if err := s.Initialize(); err != nil {
		t.Fatal(err)
	}
	defer s.Close()
	f(s)
}

func TestLensOverview(t *testing.T) {
	withLensStore(t, func(s *SQLiteStore) {
		seedLens(t, s)
		ov, err := s.LensOverview()
		if err != nil {
			t.Fatal(err)
		}
		if ov.Transactions != 3 {
			t.Fatalf("transactions: want 3 got %d", ov.Transactions)
		}
		// volume_by_asset: DZD.2 = 1000+400 = 1400, EUR.2 = 50
		vol := map[string]int64{}
		for _, v := range ov.VolumeByAsset {
			vol[v.Asset] = v.Total
		}
		if vol["DZD.2"] != 1400 || vol["EUR.2"] != 50 {
			t.Fatalf("volume_by_asset wrong: %+v", ov.VolumeByAsset)
		}
		if len(ov.TopAccounts) == 0 {
			t.Fatal("expected top accounts")
		}
		// @client:a DZD.2 volume = 1000 (in) + 400 (out) = 1400, the top entry
		if ov.TopAccounts[0].Account != "@client:a" || ov.TopAccounts[0].Asset != "DZD.2" || ov.TopAccounts[0].Volume != 1400 {
			t.Fatalf("top account wrong: %+v", ov.TopAccounts[0])
		}
	})
}

func TestLensFlows(t *testing.T) {
	withLensStore(t, func(s *SQLiteStore) {
		seedLens(t, s)
		edges, err := s.LensFlows(100)
		if err != nil {
			t.Fatal(err)
		}
		// find the @world->@client:a DZD.2 edge on 2026-07-01
		var found *FlowEdgeAlias
		for i := range edges {
			e := edges[i]
			if e.Source == "@world" && e.Destination == "@client:a" && e.Asset == "DZD.2" && e.TimeBucket == "2026-07-01" {
				found = &FlowEdgeAlias{e.Amount, e.Count}
			}
		}
		if found == nil || found.Amount != 1000 || found.Count != 1 {
			t.Fatalf("expected world->client:a DZD.2 2026-07-01 amount=1000 count=1, edges=%+v", edges)
		}
	})
}

	withLensStore(t, func(s *SQLiteStore) {
		seedLens(t, s)

  - `LensOverview()`: `CountTransactions()` (reuse), `CountAccounts()` (reuse); `SELECT asset, SUM(amount) FROM postings GROUP BY asset` → VolumeByAsset; top accounts:
    ```sql
    SELECT account, asset, SUM(amount) AS vol FROM (
      SELECT source AS account, asset, amount FROM postings
      UNION ALL
      SELECT destination AS account, asset, amount FROM postings
    ) GROUP BY account, asset ORDER BY vol DESC LIMIT 10
    ```
  - `LensFlows(limit)`:
    ```sql
    SELECT p.source, p.destination, p.asset, substr(t.timestamp,1,10) AS time_bucket,
           SUM(p.amount) AS amount, COUNT(*) AS cnt
    FROM postings p JOIN transactions t ON p.txid = t.id
    GROUP BY p.source, p.destination, p.asset, time_bucket
    ORDER BY amount DESC LIMIT ?
    ```
  - `LensRollup()`: `SELECT state, COUNT(*) FROM sharia_contracts GROUP BY state` and `SELECT action, COUNT(*) FROM guard_events GROUP BY action`, into the two maps (init non-nil maps).
  - `LensTimeSeries(account, asset)`:
    ```sql
    SELECT substr(t.timestamp,1,10) AS time_bucket,
           SUM(CASE WHEN p.destination = ? THEN p.amount ELSE 0 END) AS inn,
           SUM(CASE WHEN p.source = ? THEN p.amount ELSE 0 END) AS outt
    FROM postings p JOIN transactions t ON p.txid = t.id
    WHERE (p.source = ? OR p.destination = ?) AND p.asset = ?
    GROUP BY time_bucket ORDER BY time_bucket ASC
    ```
  Use raw `s.db.Query` with `?` placeholders (sqlite) — simpler than go-sqlbuilder for the UNION/CASE. Always init map fields so empty ledgers return `{}` not nil.

```go
package actions

import (
	"net/http"

	"github.com/amezianechayer/corren/ledger"
	"github.com/gin-gonic/gin"
)

type LensController struct {
	BaseController
}

func NewLensController() LensController { return LensController{} }

func (ctl *LensController) store(c *gin.Context) *ledger.Ledger {
	l, _ := c.Get("ledger")
	return l.(*ledger.Ledger)
}

func (ctl *LensController) Overview(c *gin.Context) {
	ov, err := ctl.store(c).Store().LensOverview()
	if err != nil {
		ctl.responseError(c, http.StatusInternalServerError, err)
		return
	}
	ctl.response(c, http.StatusOK, ov)
}

func (ctl *LensController) Flows(c *gin.Context) {
	limit := 100
	if v := c.Query("limit"); v != "" {
		if n, err := strconvAtoiClamped(v, 1, 1000); err == nil {
			limit = n
		}
	}
	edges, err := ctl.store(c).Store().LensFlows(limit)
	if err != nil {
		ctl.responseError(c, http.StatusInternalServerError, err)
		return
	}
	ctl.response(c, http.StatusOK, edges)
}

func (ctl *LensController) Rollup(c *gin.Context) {
	r, err := ctl.store(c).Store().LensRollup()
	if err != nil {
		ctl.responseError(c, http.StatusInternalServerError, err)
		return
	}
	ctl.response(c, http.StatusOK, r)
}

func (ctl *LensController) TimeSeries(c *gin.Context) {
	account := c.Query("account")
	asset := c.Query("asset")
	if account == "" || asset == "" {
		ctl.responseError(c, http.StatusBadRequest, errLensParams)
		return
	}
	ts, err := ctl.store(c).Store().LensTimeSeries(account, asset)
	if err != nil {
		ctl.responseError(c, http.StatusInternalServerError, err)
		return
	}
	ctl.response(c, http.StatusOK, ts)
}
```

Add helpers at the bottom of the file:
```go
var errLensParams = &lensError{"account and asset query params are required"}

type lensError struct{ msg string }

func (e *lensError) Error() string { return e.msg }

func strconvAtoiClamped(s string, min, max int) (int, error) {
	n, err := strconv.Atoi(s)
	if err != nil {
		return 0, err
	}
	if n < min {
		n = min
	}
	if n > max {
		n = max
	}
	return n, nil
}
```
and import `"strconv"`.
